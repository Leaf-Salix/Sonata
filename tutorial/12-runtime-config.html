  • Python conftest.py 的基本概念（pytest 的配置文件）

如果你不熟悉这些概念，建议先阅读上述教程。

提问：什么是 RUNTIME_CONFIG？

RUNTIME_CONFIG 是 PyPTO 的运行时配置字典，定义在 kernel_config.py 中。 它包含三个核心字段：

RUNTIME_CONFIG = {
    "runtime": "sim",           # 运行时类型
    "block_dim": 32,            # 块维度
    "aicpu_thread_num": 4,      # AICPU 线程数
}

PyPTO 的运行时系统（simpler）在执行编译产物时会读取这个配置。

提问：Sonata 如何注入 RUNTIME_CONFIG？

在 v0.22 中，Sonata 可以将分析结果注入到 RUNTIME_CONFIG["sonata"] 中：

RUNTIME_CONFIG = {
    "runtime": "sim",
    "block_dim": 32,
    "aicpu_thread_num": 4,
    "sonata": {                          # Sonata 注入的字段
        "schema_version": 1,
        "eligible": True,
        "task_count": 5,
        "suggested_block_dim": 64,
        "suggested_aicpu_thread_num": 8,
        "memory_peak_bytes": 1024000,
        "region_statuses": {
            "region_0": "static",
            "region_1": "dynamic",
        },
        "guard_count": 2,
        "guard_symbols": ["batch_size", "seq_len"],
    },
}
关键点：Sonata 不修改 PyPTO 的 _generate_config_file() 函数。 而是在编译后，通过 conftest hook 将 sonata 字段注入到已生成的 kernel_config.py 中。

提问：为什么使用 repr() 而不是 json.dumps()？

kernel_config.py 是 Python 源代码，通过 importlib 加载。 因此必须使用 Python 原生字面量，而不是 JSON：

方法输出是否有效 Python
json.dumps({"eligible": True})'{"eligible": true}'❌ true 不是 Python
repr({"eligible": True})"{'eligible': True}"✅ True 是 Python
注意：如果使用 json.dumps()，kernel_config.py 会报错： NameError: name 'true' is not defined。

提问：SonataRuntimeConfig 是什么？

SonataRuntimeConfig 是 Sonata 分析结果的数据类，定义在 runtime_config.py 中：

@dataclass(frozen=True)
class SonataRuntimeConfig:
    eligible: bool
    task_count: int
    suggested_block_dim: int | None = None
    suggested_aicpu_thread_num: int | None = None
    memory_peak_bytes: int | None = None
    region_statuses: dict[str, str] = field(default_factory=dict)
    guard_count: int = 0
    guard_symbols: tuple[str, ...] = ()
    schema_version: int = SONATA_RUNTIME_CONFIG_SCHEMA_VERSION

它有两个关键方法：

  • to_run_config_dict() — 转换为 Python 字典，用于注入 kernel_config.py
  • from_dict() — 从字典加载，用于读取 RUNTIME_CONFIG["sonata"]

提问：如何使用 SonataRuntimeConfig？

创建 SonataRuntimeConfig

from sonata.runtime_config import SonataRuntimeConfig

# 创建配置
config = SonataRuntimeConfig(
    eligible=True,
    task_count=5,
    suggested_block_dim=64,
    suggested_aicpu_thread_num=8,
    memory_peak_bytes=1024000,
    region_statuses={"region_0": "static", "region_1": "dynamic"},
    guard_count=2,
    guard_symbols=("batch_size", "seq_len"),
)

# 转换为字典
config_dict = config.to_run_config_dict()
print(config_dict)
# {'schema_version': 1, 'eligible': True, 'task_count': 5, ...}

从字典加载

from sonata.runtime_config import SonataRuntimeConfig

# 从 RUNTIME_CONFIG["sonata"] 加载
sonata_raw = RUNTIME_CONFIG.get("sonata")
if sonata_raw:
    config = SonataRuntimeConfig.from_dict(sonata_raw)
    if config and config.eligible:
        print(f"Eligible: {config.task_count} tasks")
        print(f"Block dim: {config.suggested_block_dim}")

提问：如何注入 RUNTIME_CONFIG？

注入过程由 st_sonata conftest 的 compile hook 完成：

def _patch_kernel_config_sonata(work_dir: Path, sonata_dict: dict) -> None:
    """Inject sonata dict into RUNTIME_CONFIG in kernel_config.py."""
    config_path = work_dir / "kernel_config.py"
    if not config_path.exists():
        return

    content = config_path.read_text(encoding="utf-8")
    marker = 'RUNTIME_CONFIG'
    idx = content.find(marker)
    if idx < 0:
        return

    # 找到 RUNTIME_CONFIG 块的闭合括号
    brace_count = 0
    close_idx = -1
    for i in range(idx, len(content)):
        if content[i] == "{":
            brace_count += 1
        elif content[i] == "}":
            brace_count -= 1
            if brace_count == 0:
                close_idx = i
                break

    if close_idx < 0:
        return

    # 使用 repr() 生成 Python 原生字面量
    sonata_repr = repr(sonata_dict)
    sonata_line = f'\t"sonata": {sonata_repr},\n'

    # 插入到 RUNTIME_CONFIG 块中
    new_content = content[:close_idx] + sonata_line + content[close_idx:]

    # 原子写入：先写临时文件，再重命名
    import tempfile
    fd, tmp_path = tempfile.mkstemp(
        dir=str(work_dir), prefix="kernel_config_", suffix=".py.tmp"
    )
    try:
        os.write(fd, new_content.encode("utf-8"))
        os.close(fd)
        os.replace(tmp_path, str(config_path))
    except BaseException:
        if fd >= 0:
            os.close(fd)
        os.unlink(tmp_path)
        raise

提问：如何读取 RUNTIME_CONFIG？

读取过程由 apply_sonata_runtime_hints() 完成。这个函数返回一个 SonataRuntimeHints 对象， 包含以下字段：

  • block_dim：块维度（可能被 Sonata 建议值修改）
  • aicpu_thread_num：AICPU 线程数（可能被 Sonata 建议值修改）
  • sonata_applied：是否成功应用了 Sonata 建议
  • reason：应用/未应用的原因（用于调试）

_unchanged() 是一个辅助函数，用于在无法应用 Sonata 建议时返回原始值：

def _unchanged(block_dim, aicpu_thread_num, reason):
    """返回未修改的配置，记录原因。"""
    return SonataRuntimeHints(
        block_dim=block_dim,
        aicpu_thread_num=aicpu_thread_num,
        sonata_applied=False,
        reason=reason,
    )

完整的读取过程：

def _apply_from_kernel_config(config_path, block_dim, aicpu_thread_num, user_block_dim):
    """Read RUNTIME_CONFIG["sonata"] from kernel_config.py."""
    import importlib.util

    # 加载 kernel_config.py 模块
    spec = importlib.util.spec_from_file_location("_kc", str(config_path))
    if spec is None or spec.loader is None:
        return _unchanged(block_dim, aicpu_thread_num, "config_load_failed")

    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)

    # 读取 RUNTIME_CONFIG["sonata"]
    sonata_raw = getattr(mod, "RUNTIME_CONFIG", {}).get("sonata")
    if not sonata_raw:
        return _unchanged(block_dim, aicpu_thread_num, "no_sonata_key")

    # 解析 SonataRuntimeConfig
    from .runtime_config import SonataRuntimeConfig
    sonata_cfg = SonataRuntimeConfig.from_dict(sonata_raw)
    if sonata_cfg is None:
        return _unchanged(block_dim, aicpu_thread_num, "invalid_sonata_config")

    if not sonata_cfg.eligible:
        return _unchanged(block_dim, aicpu_thread_num, "plan_not_eligible")

    # 用户显式指定 block_dim 时不覆盖
    if user_block_dim is not None:
        return _unchanged(block_dim, aicpu_thread_num, "user_supplied_block_dim")

    # 应用建议值
    new_block_dim = sonata_cfg.suggested_block_dim or block_dim
    new_aicpu = sonata_cfg.suggested_aicpu_thread_num or aicpu_thread_num

    return SonataRuntimeHints(
        block_dim=new_block_dim,
        aicpu_thread_num=new_aicpu,
        sonata_applied=True,
        reason="runtime_config_sonata",
    )

提问：双源读取策略是什么？

apply_sonata_runtime_hints() 使用双源读取策略：

  1. 优先读取 sonata_plan.json — 包含完整数据（Score、PlanHandle 等）
  2. 回退到 RUNTIME_CONFIG["sonata"] — 只包含调度提示（block_dim、aicpu_thread_num）
def _do_apply(work_dir, block_dim, aicpu_thread_num, user_block_dim):
    plan_path = Path(work_dir) / "sonata_plan.json"
    config_path = Path(work_dir) / "kernel_config.py"

    # 策略 1: 读取 sonata_plan.json（优先 — 完整数据）
    if plan_path.exists():
        return _apply_from_plan_json(...)

    # 策略 2: 读取 RUNTIME_CONFIG["sonata"]
    if config_path.exists():
        return _apply_from_kernel_config(...)

    # 没有 Sonata 数据
    return SonataRuntimeHints(
        block_dim=block_dim,
        aicpu_thread_num=aicpu_thread_num,
        sonata_applied=False,
        reason="no_sonata_data",
    )
设计理由：sonata_plan.json 包含完整的分析结果，适合需要详细信息的场景。 RUNTIME_CONFIG["sonata"] 只包含调度提示，适合轻量级场景。 双源读取确保在任何情况下都能获取 Sonata 分析结果。

def test_from_dict_roundtrip():
    """Test from_dict round-trip preserves data."""
    config = SonataRuntimeConfig(
        eligible=True,
        task_count=5,
        suggested_block_dim=64,
    )
    d = config.to_run_config_dict()
    restored = SonataRuntimeConfig.from_dict(d)
    assert restored is not None
    assert restored.eligible == config.eligible
    assert restored.task_count == config.task_count
    assert restored.suggested_block_dim == config.suggested_block_dim


def test_from_dict_invalid_type():
    """Test from_dict returns None for non-dict."""
    assert SonataRuntimeConfig.from_dict("not a dict") is None
    assert SonataRuntimeConfig.from_dict(42) is None
    assert SonataRuntimeConfig.from_dict(None) is None


def test_from_dict_unknown_schema_version():

提问：当前有什么限制？

限制：
  • RUNTIME_CONFIG["sonata"] 只包含调度提示，不包含完整的 Score 数据
  • Guard 元数据（guard_count、guard_symbols）仅用于信息/日志，不执行运行时 guard 评估
  • 如果 kernel_config.py 被重新生成，Sonata 注入会丢失（但 conftest hook 会重新注入）
  • 原子写入使用临时文件+重命名，但不是所有文件系统都支持原子重命名
总结：RUNTIME_CONFIG 注入是 v0.22 的核心功能，使得 Sonata 分析结果可以嵌入到 PyPTO 的 kernel_config.py 中。通过 SonataRuntimeConfig 数据类和 repr() 注入，实现了与 PyPTO 运行时系统的无缝集成。
